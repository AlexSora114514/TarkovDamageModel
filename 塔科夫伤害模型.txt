        { name: "6B23-2 ballistic plate (Back)", class: 4, bt: 0.18, maxDur: 76, mat: "装甲钢", pos: "后", weightClass: "heavy" },
        { name: "AR500 Legacy Plate ballistic plate", class: 3, bt: 0.18, maxDur: 120, mat: "复合材料", pos: "前后", weightClass: "heavy" },
        { name: "Zhuk-3 ballistic plate (Front)", class: 3, bt: 0.24, maxDur: 118, mat: "超高分子聚乙烯", pos: "前", weightClass: "light" },
        { name: "PRTCTR Lightweight ballistic plate", class: 3, bt: 0.26, maxDur: 103, mat: "超高分子聚乙烯", pos: "前后", weightClass: "light" },
        { name: "6B12 ballistic plates (Front)", class: 3, bt: 0.18, maxDur: 95, mat: "装甲钢", pos: "前", weightClass: "heavy" },

    // ================= 核心计算 =================
    const THORAX_HP = 85;

    const calcArmorDefense = (durPct, armorClass) => (121 - 5000 / (45 + durPct * 2)) * armorClass * 0.1;

    function calcEffectiveClass(durPct, armorClass) {
        const a = calcArmorDefense(durPct, armorClass);
        const aOne = Math.floor(a * 10) / 10;
        const intPart = Math.floor(aOne / 10);
        const decPart = aOne - intPart * 10;
        const eff = intPart + (decPart > 0 ? 1 : 0);
        return Math.min(armorClass, Math.max(1, eff));
    }

    function calcPenChance(pen, a) {
        if (pen <= a - 15) return 0;
        if (pen < a) return Math.min(1, 0.4 * (a - pen - 15) ** 2 / 100);
        const denom = 0.9 * a - pen;
        if (Math.abs(denom) < 1e-9) return 1;
        return Math.max(0, Math.min(1, (100 + pen / denom) / 100));
    }

    const calcPenetratingDamage = (pen, a, dmg, pellets) =>
        dmg * pellets * Math.max(0.6, Math.min(1, pen / (a + 12.5)));

    function calcArmorLoss(armorDmg, matCoef, penetrated, pen, a, pellets) {
        const base = armorDmg * pellets * matCoef;
        if (!penetrated) return base;
        const overflow = Math.min(1, Math.max(0, (pen - a) / (2 * a)));
        return base * (0.8 + 0.1 * overflow);
    }

    function calcDoubleLayerExpected(bullet, armor, plateDurPct, linerDurPct, skillReduction, ricochetChance) {
        const pellets = bullet.pellets || 1;
        const plateBroken = plateDurPct <= 0;
        const linerBroken = linerDurPct <= 0;
        const group = bulletsByCal[bullet.cal] || [];
        const fixedBlunt = calcFixedBlunt(bullet, armor, group, skillReduction);

        if (linerBroken) {
            return {
                finalDmg: bullet.dmg * pellets * (1 - ricochetChance),
                plateLoss: 0, linerLoss: 0, penChance: 1,
                penDmg: bullet.dmg * pellets, bluntDmg: 0, plateA: 0, linerA: 0
            };
        }

        const linerA = calcArmorDefense(linerDurPct, LINER.class);

        if (plateBroken) {
            const penChance = calcPenChance(bullet.pen, linerA);
            const penDmg = calcPenetratingDamage(bullet.pen, linerA, bullet.dmg, pellets);
            const linerLoss = penChance * calcArmorLoss(bullet.armorDmg, LINER.matDamage, true, bullet.pen, linerA, pellets)
                            + (1 - penChance) * calcArmorLoss(bullet.armorDmg, LINER.matDamage, false, bullet.pen, linerA, pellets);
            return {
                finalDmg: penChance * penDmg * (1 - ricochetChance),
                plateLoss: 0, linerLoss, penChance, penDmg, bluntDmg: 0, plateA: 0, linerA
            };
        }

        const plateA = calcArmorDefense(plateDurPct, armor.class);
        const platePenChance = calcPenChance(bullet.pen, plateA);
        const platePenDmg = calcPenetratingDamage(bullet.pen, plateA, bullet.dmg, pellets);
        const penAfter = bullet.pen * getPenDecay(armor.mat);
        const linerPenChance = calcPenChance(penAfter, linerA);
        const linerPenDmg = calcPenetratingDamage(penAfter, linerA, platePenDmg, 1);
        const linerExpected = linerPenChance * linerPenDmg;

        const finalDmg = (platePenChance * linerExpected + (1 - platePenChance) * fixedBlunt) * (1 - ricochetChance);

        const plateLoss = platePenChance * calcArmorLoss(bullet.armorDmg, getMatDamage(armor.mat), true, bullet.pen, plateA, pellets)
                        + (1 - platePenChance) * calcArmorLoss(bullet.armorDmg, getMatDamage(armor.mat), false, bullet.pen, plateA, pellets);

        const linerLoss = platePenChance * (linerPenChance * calcArmorLoss(bullet.armorDmg, LINER.matDamage, true, penAfter, linerA, pellets)
                                          + (1 - linerPenChance) * calcArmorLoss(bullet.armorDmg, LINER.matDamage, false, penAfter, linerA, pellets));

        return { finalDmg, plateLoss, linerLoss, penChance: platePenChance, penDmg: linerPenDmg, bluntDmg: fixedBlunt, plateA, linerA };
    }

    // ================= UI 初始化 =================
    const $ = id => document.getElementById(id);
    const bulletSelect = $('bulletSelect'), armorSelect = $('armorSelect'), weaponSelect = $('weaponSelect');
    const durabilityInput = $('durability'), durabilityValue = $('durabilityValue');
    const customRpm = $('customRpm'), rpmValue = $('rpmValue');
    const armorSkillInput = $('armorSkill'), armorSkillValue = $('armorSkillValue');
    const simTbody = document.querySelector('#simTable tbody');

    let currentCal = "";
    bullets.forEach((b, i) => {
        if (b.cal !== currentCal) {
            currentCal = b.cal;
            const grp = document.createElement('option');
            grp.disabled = true;
            grp.textContent = `── ${b.cal} ──`;
            bulletSelect.appendChild(grp);
        }
        const opt = document.createElement('option');
        opt.value = i;
        opt.textContent = `${b.name} (穿:${b.pen} 肉:${b.dmg} 甲:${b.armorDmg})`;
        bulletSelect.appendChild(opt);
    });

    armors.forEach((a, i) => {
        const opt = document.createElement('option');
        opt.value = i;
        opt.textContent = `[${a.class}] ${a.name} (${a.weightClass === "heavy" ? "重甲" : "轻甲"}-${a.pos}-钝伤${(a.bt * 100).toFixed(0)}%-等效耐久${a.maxDur}-${a.mat})`;
        armorSelect.appendChild(opt);
        if (a.name === "Cult Termite ballistic plate") armorSelect.value = i;
    });

    function updateWeapons(cal) {
        weaponSelect.innerHTML = '';
        const weapons = weaponsByCaliber[cal] || [];
        if (!weapons.length) {
            const opt = document.createElement('option');
            opt.value = 30; opt.textContent = '无对应武器';
            weaponSelect.appendChild(opt);
            customRpm.value = 30; rpmValue.textContent = 30;
            return;
        }
        weapons.sort((a, b) => b.rpm - a.rpm);
        weapons.forEach(w => {
            const opt = document.createElement('option');
            opt.value = w.rpm; opt.textContent = `${w.name} (${w.rpm} 发/分)`;
            weaponSelect.appendChild(opt);
        });
        customRpm.value = weapons[0].rpm;
        rpmValue.textContent = weapons[0].rpm;
    }

    const getSkillReduction = (wc, lv) => wc === "heavy" ? (lv >= 51 ? 0.20 : lv * 0.004) : 0;
    const getRicochetChance = (wc, lv) => (wc === "heavy" && lv >= 51) ? 0.20 : 0;

    function update() {
        const bullet = bullets[bulletSelect.value];
        const armor = armors[armorSelect.value];
        const currentDurPct = parseInt(durabilityInput.value);
        const rpm = parseInt(customRpm.value);
        const skillLevel = parseInt(armorSkillInput.value);
        const linerDurPct = 100;

        if (!bullet || !armor) return;

        const skillReduction = getSkillReduction(armor.weightClass, skillLevel);
        const ricochetChance = getRicochetChance(armor.weightClass, skillLevel);

        $('armorTypeDisplay').textContent = armor.weightClass === "heavy" ? "重甲" : "轻甲";
        $('bluntReductionDisplay').textContent = `${(skillReduction * 100).toFixed(1)}%`;
        $('ricochetRow').style.display = (armor.weightClass === "heavy" && skillLevel >= 51) ? "flex" : "none";

        const result = calcDoubleLayerExpected(bullet, armor, currentDurPct, linerDurPct, skillReduction, ricochetChance);

        $('penChance').textContent = `${(result.penChance * 100).toFixed(1)}%`;
        $('finalDamage').textContent = result.penDmg.toFixed(1);
        $('bluntDamage').textContent = result.bluntDmg.toFixed(1);
        $('expectedDamage').textContent = result.finalDmg.toFixed(1);
        $('armorLoss').textContent = (result.plateLoss + result.linerLoss).toFixed(2);
        $('remainingDurability').textContent = Math.max(0, currentDurPct - (result.plateLoss / armor.maxDur) * 100).toFixed(1) + '%';
        $('effClass').textContent = currentDurPct <= 0 ? `${LINER.class}（甲皮）` : calcEffectiveClass(currentDurPct, armor.class);
        $('effDefense').textContent = currentDurPct <= 0 ? calcArmorDefense(linerDurPct, LINER.class).toFixed(1) : result.plateA.toFixed(1);
        $('pelletCount').textContent = bullet.pellets || 1;

        simTbody.innerHTML = '';
        let simPlateDur = currentDurPct, simLinerDur = linerDurPct, thoraxHP = THORAX_HP, shotNum = 0;
        const maxShots = 100;
        const fireIntervalMs = 60000 / rpm;

        while (thoraxHP > 0 && shotNum < maxShots) {
            shotNum++;
            const r = calcDoubleLayerExpected(bullet, armor, simPlateDur, simLinerDur, skillReduction, ricochetChance);
            thoraxHP = Math.max(0, thoraxHP - r.finalDmg);
            simPlateDur = Math.max(0, simPlateDur - (r.plateLoss / armor.maxDur) * 100);
            simLinerDur = Math.max(0, simLinerDur - (r.linerLoss / LINER.maxDur) * 100);

            const plateDisplay = simPlateDur <= 0 ? '<span class="broken">已击穿</span>' : `${simPlateDur.toFixed(1)}%`;
            const linerDisplay = simLinerDur <= 0 ? '<span class="broken">已击穿</span>' : `${simLinerDur.toFixed(1)}%`;
            const hpDisplay = thoraxHP <= 0 ? '<span style="color:var(--red);font-weight:bold;">0.0</span>' : thoraxHP.toFixed(1);
            const row = document.createElement('tr');
            row.innerHTML = `
                <td>${shotNum}</td>
                <td>${plateDisplay}</td>
                <td>${linerDisplay}</td>
                <td>${r.plateA.toFixed(1)}</td>
                <td>${r.linerA.toFixed(1)}</td>
                <td>${(r.penChance * 100).toFixed(0)}%</td>
                <td>${r.penDmg.toFixed(1)}</td>
                <td>${r.bluntDmg.toFixed(1)}</td>
                <td style="color:var(--yellow);font-weight:bold;">${r.finalDmg.toFixed(1)}</td>
                <td>${r.plateLoss.toFixed(2)}</td>
                <td>${r.linerLoss.toFixed(2)}</td>
                <td>${hpDisplay}</td>
                <td style="color:var(--accent);">${((shotNum - 1) * fireIntervalMs).toFixed(0)}</td>
            `;
            simTbody.appendChild(row);
        }
    }

    // ================= TTK 排行榜（仅全自动，每种子弹取射速前二且射速去重，前200） =================
    function calcTTK(bullet, armor, rpm) {
        const skillReduction = 0.10;
        const ricochetChance = 0;
        const linerDurPct = 100;
        let plateDur = 100, thoraxHP = THORAX_HP, shotNum = 0;
        const maxShots = 100;
        while (thoraxHP > 0 && shotNum < maxShots) {
            shotNum++;
            const r = calcDoubleLayerExpected(bullet, armor, plateDur, linerDurPct, skillReduction, ricochetChance);
            thoraxHP = Math.max(0, thoraxHP - r.finalDmg);
            plateDur = Math.max(0, plateDur - (r.plateLoss / armor.maxDur) * 100);
            if (thoraxHP <= 0) break;
        }
        return { ttk: (shotNum - 1) * (60000 / rpm), shots: shotNum };
    }

    function buildRanking() {
        const titaniumPlate = armors.find(a => a.name === "Cult Termite ballistic plate");
        if (!titaniumPlate) return [];

        const results = [];
        for (const bullet of bullets) {
            const autoWeapons = (weaponsByCaliber[bullet.cal] || []).filter(w => w.auto !== false);
            // 按 RPM 降序，去重相同射速（保留第一个），取前 2
            const seenRpm = new Set();
            const picked = [];
            for (const w of autoWeapons.sort((a, b) => b.rpm - a.rpm)) {
                if (seenRpm.has(w.rpm)) continue;
                seenRpm.add(w.rpm);
                picked.push(w);
                if (picked.length >= 2) break;
            }
            for (const weapon of picked) {
                const { ttk, shots } = calcTTK(bullet, titaniumPlate, weapon.rpm);
                if (shots >= 100) continue;
                results.push({ bullet, weapon, ttk, shots });
            }
        }
        results.sort((a, b) => a.ttk - b.ttk);
        return results.slice(0, 200);
    }

    function renderRanking() {
        const data = buildRanking();
        const tbody = document.querySelector('#rankTable tbody');
        tbody.innerHTML = '';
        data.forEach((d, i) => {
            const row = document.createElement('tr');
            row.innerHTML = `
                <td>${i + 1}</td>
                <td>${d.bullet.cal}</td>
                <td>${d.bullet.name || '—'}</td>
                <td>${d.weapon.name}</td>
                <td>${d.weapon.rpm}</td>
                <td>${d.bullet.pen}</td>
                <td>${d.bullet.dmg}</td>
                <td style="color:var(--accent);font-weight:bold;">${d.ttk.toFixed(0)}</td>
                <td>${d.shots}</td>
            `;
            tbody.appendChild(row);
        });
    }

    armorSkillInput.addEventListener('input', e => { armorSkillValue.textContent = e.target.value === "51" ? "精英" : e.target.value; update(); });
    armorSelect.addEventListener('change', update);
    durabilityInput.addEventListener('input', e => { durabilityValue.textContent = e.target.value + '%'; update(); });

    const initBullet = bullets[bulletSelect.value];
    if (initBullet) updateWeapons(initBullet.cal);
    update();
